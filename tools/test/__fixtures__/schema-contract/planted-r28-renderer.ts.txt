// Planted (SC R28): a renderer with its own format list and its own amount regex.
export const STATEMENT_FORMATS = [
  { name: 'brackets', negative: 'brackets', thousands: ',' },
  { name: 'plain', negative: 'leading', thousands: ',' },
]
const AMOUNT = /^-?\d{1,3}(,\d{3})*\.\d{2}$/
export function looksLikeAmount(text: string): boolean {
  return AMOUNT.test(text)
}
